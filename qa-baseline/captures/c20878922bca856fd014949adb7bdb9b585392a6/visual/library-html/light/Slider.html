Usage
import { Slider, SLIDER_SIZE } from "gwan-design-system";
import { useState } from "react";

// Single thumb
const SingleExample = () => {
  const [vol, setVol] = useState(40);
  return (
    <Slider
      label="Volume"
      value={vol}
      onChange={(v) => setVol(v as number)}
      min={0}
      max={100}
      step={1}
      showTooltip
      size={SLIDER_SIZE.MD}   // SM | MD | LG
    />
  );
};

// Dual thumb (range)
const RangeExample = () => {
  const [range, setRange] = useState<[number, number]>([20, 80]);
  return (
    <Slider
      label="Price range"
      value={range}
      onChange={(v) => setRange(v as [number, number])}
      min={0}
      max={200}
      step={5}
      showTooltip
    />
  );
};
