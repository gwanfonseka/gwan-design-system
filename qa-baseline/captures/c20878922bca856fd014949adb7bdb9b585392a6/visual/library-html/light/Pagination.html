Usage
import { Pagination, IPaging } from "gwan-design-system";

const Example = () => {
  const [paging, setPaging] = useState<IPaging>({
    total: 112,
    page: 1,
    size: 10,
  });

   return (
     <Pagination
       page={paging.page}
       size={paging.size}
       total={paging.total}
       options={[
         { label: "10" },
         { label: "25" },
         { label: "50" },
         { label: "100" },
         { label: "200" },
       ]}
       onChange={(paging: IPaging) => setPaging(paging)}
       className="custom-class" // optional, you can add your own styles
     />
   );
};
