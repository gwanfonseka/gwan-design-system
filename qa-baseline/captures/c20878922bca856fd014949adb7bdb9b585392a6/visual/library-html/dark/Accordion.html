Usage
import { Accordion, ACCORDION_VARIANT } from "gwan-design-system";

const items = [
  {
    key: "q1",
    title: "What is GWAN?",
    content: "A production-ready component library for Next.js.",
  },
  {
    key: "q2",
    title: "How do I install it?",
    content: "Run: npm install gwan-design-system",
  },
  {
    key: "q3",
    title: "Disabled item",
    content: "Not accessible.",
    disabled: true,
  },
];

const Example = () => (
  <Accordion
    items={items}
    defaultOpenKey="q1"
    variant={ACCORDION_VARIANT.DEFAULT} // default | bordered | flush
    allowMultiple={false}
  />
);
