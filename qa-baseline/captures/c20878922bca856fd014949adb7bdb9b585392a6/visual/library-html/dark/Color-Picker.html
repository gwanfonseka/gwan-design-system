Usage
import { ColorPicker, FORM_ELEMENT_SIZE } from "gwan-design-system";
import { useState } from "react";

const Example = () => {
  const [color, setColor] = useState("#3b82f6");

  return (
    <>
      <ColorPicker
        value={color}
        onChange={setColor}
        label="Pick a color"
        size={FORM_ELEMENT_SIZE.MD}    // MD | SM
      />

      {/* Custom presets */}
      <ColorPicker
        value={color}
        onChange={setColor}
        presets={["#ef4444", "#22c55e", "#3b82f6", "#8b5cf6"]}
      />

      {/* Disabled */}
      <ColorPicker value="#3b82f6" onChange={() => {}} disabled />
    </>
  );
};
